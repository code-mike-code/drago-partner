import type { Language } from '@/hooks/useLanguage'

export const SITE_URL = 'https://dragopartner.pl'

export function headingId(text: string): string {
  return text
    .replace(/[łŁ]/g, 'l')
    .normalize('NFD')
    .replace(/[̀-ͯ]/g, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '')
}

const DATE_LOCALES: Record<Language, string> = {
  pl: 'pl-PL',
  en: 'en-GB',
  ua: 'uk-UA',
  sr: 'sr-Latn-RS',
  gr: 'ka-GE',
}

export function formatDate(iso: string, language: Language): string {
  return new Intl.DateTimeFormat(DATE_LOCALES[language], {
    dateStyle: 'long',
    timeZone: 'UTC',
  }).format(new Date(`${iso}T00:00:00Z`))
}

`src/content/blog/validate.ts`:

```ts
import type { Article } from './types'
import { headingId } from './utils'

const SLUG_RE = /^[a-z0-9]+(?:-[a-z0-9]+)*$/
const DATE_RE = /^\d{4}-\d{2}-\d{2}$/
const BLOG_LINK_RE = /\]\(\/blog\/([a-z0-9-]+)\/?\)/g

export function validateArticles(articles: Article[]): string[] {
  const errors: string[] = []
  const slugs = new Set(articles.map((a) => a.slug))
  const seen = new Set<string>()

  for (const a of articles) {
    const at = `[${a.slug}]`
    if (!SLUG_RE.test(a.slug)) errors.push(`${at} invalid slug`)
    if (seen.has(a.slug)) errors.push(`${at} duplicate slug`)
    seen.add(a.slug)

    if (a.title.length > 70) errors.push(`${at} title is ${a.title.length} chars (max 70)`)
    if (a.description.length < 70 || a.description.length > 160) {
      errors.push(`${at} description is ${a.description.length} chars (need 70-160)`)
    }
    if (!DATE_RE.test(a.publishedAt) || !DATE_RE.test(a.updatedAt)) {
      errors.push(`${at} dates must be YYYY-MM-DD`)
    } else if (a.updatedAt < a.publishedAt) {
      errors.push(`${at} updatedAt is before publishedAt`)
    }
    if (a.readingMinutes < 1) errors.push(`${at} readingMinutes must be >= 1`)
    if (!a.image || !a.imageAlt) errors.push(`${at} image and imageAlt are required`)

    const h2s = a.blocks.flatMap((b) => (b.type === 'h2' ? [b.text] : []))
    if (h2s.length < 3) errors.push(`${at} needs at least 3 h2 headings (has ${h2s.length})`)
    const ids = h2s.map(headingId)
    if (new Set(ids).size !== ids.length) errors.push(`${at} duplicate h2 ids`)
    if (ids.some((id) => id === '')) errors.push(`${at} h2 with empty id`)

    for (const rel of a.related ?? []) {
      if (rel === a.slug || !slugs.has(rel)) errors.push(`${at} invalid related slug "${rel}"`)
    }

    const texts = a.blocks.flatMap((b) =>
      b.type === 'list' ? b.items : [b.text],
    )
    for (const text of texts) {
      for (const m of text.matchAll(BLOG_LINK_RE)) {
        if (!slugs.has(m[1])) errors.push(`${at} link to unknown article "${m[1]}"`)
      }
    }

    ;(a.faq ?? []).forEach((f, i) => {
      if (!f.question.trim() || !f.answer.trim()) errors.push(`${at} faq[${i}] is empty`)
    })
  }
  return errors
}
```

import type { ReactNode } from 'react'
import { Link } from 'react-router-dom'
import type { Block } from '@/content/blog/types'
import { headingId } from '@/content/blog/utils'
import { cn } from '@/lib/utils'

const INLINE_RE = /\[([^\]]+)\]\(([^)]+)\)/g
const LINK_CLASS =
  'font-semibold text-dark underline decoration-illuminating decoration-2 underline-offset-4 hover:bg-illuminating transition-colors'

function renderInline(text: string): ReactNode[] {
  const nodes: ReactNode[] = []
  let last = 0
  for (const m of text.matchAll(INLINE_RE)) {
    const index = m.index ?? 0
    const [whole, label, href] = m
    if (index > last) nodes.push(text.slice(last, index))
    if (href.startsWith('/') && !href.startsWith('/#')) {
      nodes.push(
        <Link key={index} to={href} className={LINK_CLASS}>
          {label}
        </Link>,
      )
    } else {
      const external = href.startsWith('http')
      nodes.push(
        <a
          key={index}
          href={href}
          className={LINK_CLASS}
          {...(external && { target: '_blank', rel: 'noopener noreferrer' })}
        >
          {label}
        </a>,
      )
    }
    last = index + whole.length
  }
  if (last < text.length) nodes.push(text.slice(last))
  return nodes
}

export function ArticleBody({ blocks }: { blocks: Block[] }) {
  return (
    <div>
      {blocks.map((block, i) => {
        switch (block.type) {
          case 'h2':
            return (
              <h2
                key={i}
                id={headingId(block.text)}
                className="scroll-mt-24 text-2xl sm:text-3xl font-black text-dark leading-tight tracking-tight mt-14 mb-5"
              >
                {block.text}
              </h2>
            )
          case 'p':
            return (
              <p key={i} className="text-lg leading-relaxed text-dark/80 mb-6">
                {renderInline(block.text)}
              </p>
            )
          case 'list': {
            const Tag = block.ordered ? 'ol' : 'ul'
            return (
              <Tag
                key={i}
                className={cn(
                  'pl-6 mb-6 space-y-2 text-lg leading-relaxed text-dark/80',
                  block.ordered ? 'list-decimal' : 'list-disc',
                )}
              >
                {block.items.map((item, j) => (
                  <li key={j} className="pl-1">
                    {renderInline(item)}
                  </li>
                ))}
              </Tag>
            )
          }
          case 'callout':
            return (
              <aside
                key={i}
                className="border-l-4 border-illuminating bg-ghost-white p-6 mb-8 text-base leading-relaxed text-dark"
              >
                {renderInline(block.text)}
              </aside>
            )
        }
      })}
    </div>
  )
}

import { useEffect } from 'react'
import { Helmet } from 'react-helmet-async'
import { Link, useParams } from 'react-router-dom'
import { useLanguage } from '@/hooks/useLanguage'
import { Header } from '@/components/layout/Header'
import { Footer } from '@/components/layout/Footer'
import { ArticleBody } from '@/components/blog/ArticleBody'
import { CTAButton } from '@/components/ui/CTAButton'
import NotFound from '@/pages/NotFound'
import { getArticle } from '@/content/blog'
import { SITE_URL, formatDate, headingId } from '@/content/blog/utils'

export default function BlogPost() {
  const { slug = '' } = useParams()
  const { t, language, setLanguage } = useLanguage()
  const article = getArticle(slug)
  const articleLang = article?.lang

  useEffect(() => {
    if (articleLang) setLanguage(articleLang)
  }, [articleLang, setLanguage])

  if (!article) return <NotFound />

  const url = `${SITE_URL}/blog/${article.slug}/`
  const imageUrl = article.image.startsWith('http') ? article.image : `${SITE_URL}${article.image}`
  const pageTitle = `${article.title} | Drago Partner`
  const ogLocale = article.lang === 'en' ? 'en_GB' : 'pl_PL'
  const h2s = article.blocks.flatMap((b) => (b.type === 'h2' ? [b.text] : []))
  const related = (article.related ?? []).flatMap((s) => {
    const r = getArticle(s)
    return r ? [r] : []
  })

  const jsonLd: Record<string, unknown>[] = [
    {
      '@context': 'https://schema.org',
      '@type': 'Article',
      headline: article.title,
      description: article.description,
      image: imageUrl,
      datePublished: article.publishedAt,
      dateModified: article.updatedAt,
      inLanguage: article.lang,
      mainEntityOfPage: url,
      author: { '@type': 'Organization', name: 'Drago Partner', url: SITE_URL },
      publisher: {
        '@type': 'Organization',
        name: 'Drago Partner',
        logo: { '@type': 'ImageObject', url: `${SITE_URL}/logo.webp` },
      },
    },
    {
      '@context': 'https://schema.org',
      '@type': 'BreadcrumbList',
      itemListElement: [
        { '@type': 'ListItem', position: 1, name: t('blog.breadcrumbHome'), item: `${SITE_URL}/` },
        { '@type': 'ListItem', position: 2, name: 'Blog', item: `${SITE_URL}/blog/` },
        { '@type': 'ListItem', position: 3, name: article.title, item: url },
      ],
    },
  ]
  if (article.faq?.length) {
    jsonLd.push({
      '@context': 'https://schema.org',
      '@type': 'FAQPage',
      mainEntity: article.faq.map((f) => ({
        '@type': 'Question',
        name: f.question,
        acceptedAnswer: { '@type': 'Answer', text: f.answer },
      })),
    })
  }

  return (
    <>
      <Helmet>
        <title>{pageTitle}</title>
        <meta name="description" content={article.description} />
        <link rel="canonical" href={url} />
        <meta property="og:type" content="article" />
        <meta property="og:url" content={url} />
        <meta property="og:title" content={pageTitle} />
        <meta property="og:description" content={article.description} />
        <meta property="og:image" content={imageUrl} />
        <meta property="og:locale" content={ogLocale} />
        <meta property="og:site_name" content="Drago Partner" />
        <meta property="article:published_time" content={article.publishedAt} />
        <meta property="article:modified_time" content={article.updatedAt} />
        <meta name="twitter:card" content="summary_large_image" />
        <meta name="twitter:title" content={pageTitle} />
        <meta name="twitter:description" content={article.description} />
        <meta name="twitter:image" content={imageUrl} />
        {jsonLd.map((block, i) => (
          <script key={i} type="application/ld+json">
            {JSON.stringify(block)}
          </script>
        ))}
      </Helmet>

      <Header />
      <main className="pt-16 bg-white">
        <article className="py-16 md:py-24">
          <div className="max-w-3xl mx-auto px-4 sm:px-6 lg:px-8" lang={article.lang}>
            <nav aria-label="Breadcrumb" className="text-sm text-grey-mid mb-8 flex flex-wrap gap-2">
              <Link to="/" className="hover:text-dark transition-colors">
                {t('blog.breadcrumbHome')}
              </Link>
              <span aria-hidden="true">/</span>
              <Link to="/blog/" className="hover:text-dark transition-colors">
                Blog
              </Link>
            </nav>

            <h1 className="text-3xl sm:text-4xl lg:text-5xl font-black text-dark leading-tight tracking-tight">
              {article.title}
            </h1>
            <div className="w-16 h-1 bg-illuminating mt-8 mb-6" />
            <p className="text-sm text-grey-mid mb-10">
              {t('blog.published')}: {formatDate(article.publishedAt, language)}
              {article.updatedAt !== article.publishedAt &&
                ` · ${t('blog.updated')}: ${formatDate(article.updatedAt, language)}`}
              {` · ${article.readingMinutes} ${t('blog.minutes')}`}
            </p>

            <img
              src={article.image}
              alt={article.imageAlt}
              className="w-full aspect-[16/9] object-cover mb-12"
            />

            {h2s.length >= 3 && (
              <nav aria-label={t('blog.tocHeading')} className="border border-gray-200 p-6 mb-12">
                <p className="text-xs font-semibold uppercase tracking-widest text-grey-mid mb-4">
                  {t('blog.tocHeading')}
                </p>
                <ol className="space-y-2 list-decimal pl-5 text-dark">
                  {h2s.map((text) => (
                    <li key={text}>
                      <a href={`#${headingId(text)}`} className="hover:underline underline-offset-4">
                        {text}
                      </a>
                    </li>
                  ))}
                </ol>
              </nav>
            )}

            <ArticleBody blocks={article.blocks} />

            {article.faq && article.faq.length > 0 && (
              <section className="mt-16">
                <h2 className="text-2xl sm:text-3xl font-black text-dark leading-tight tracking-tight mb-8">
                  {t('blog.faqHeading')}
                </h2>
                <div className="divide-y divide-gray-200 border-y border-gray-200">
                  {article.faq.map((f) => (
                    <div key={f.question} className="py-6">
                      <h3 className="text-lg font-bold text-dark mb-2">{f.question}</h3>
                      <p className="text-dark/80 leading-relaxed">{f.answer}</p>
                    </div>
                  ))}
                </div>
              </section>
            )}

            <p className="text-sm text-grey-mid mt-12">{t('blog.disclaimer')}</p>
          </div>
        </article>

        <section className="bg-dark text-white py-16 md:py-24">
          <div className="max-w-3xl mx-auto px-4 sm:px-6 lg:px-8">
            <h2 className="text-3xl sm:text-4xl font-black leading-tight tracking-tight">
              {t('blog.ctaHeading')}
            </h2>
            <p className="text-grey-mid leading-relaxed mt-4 mb-8 max-w-xl">{t('blog.ctaText')}</p>
            <CTAButton href="/#contact-form" variant="yellow">
              {t('blog.ctaButton')}
            </CTAButton>
          </div>
        </section>

        {related.length > 0 && (
          <section className="py-16 md:py-24">
            <div className="max-w-3xl mx-auto px-4 sm:px-6 lg:px-8">
              <h2 className="text-2xl sm:text-3xl font-black text-dark leading-tight tracking-tight mb-8">
                {t('blog.relatedHeading')}
              </h2>
              <ul className="divide-y divide-gray-200 border-y border-gray-200">
                {related.map((r) => (
                  <li key={r.slug}>
                    <Link
                      to={`/blog/${r.slug}/`}
                      lang={r.lang}
                      className="block py-5 font-bold text-dark hover:text-grey-mid transition-colors"
                    >
                      {r.title}
                    </Link>
                  </li>
                ))}

```js
const ROOT_PLACEHOLDER = '<div id="root"></div>'

export function injectPage(template, { html, head, lang }) {
  if (!template.includes('</head>')) throw new Error('template has no </head>')
  if (!template.includes(ROOT_PLACEHOLDER)) throw new Error('template has no empty #root')

  return template
    .replace(/<html lang="[^"]*"/, () => `<html lang="${lang}"`)
    .replace(/<title>[\s\S]*?<\/title>\s*/, '')
    .replace(/<meta name="(?:description|keywords|twitter:[^"]*)"[^>]*>\s*/g, '')
    .replace(/<meta property="og:[^"]*"[^>]*>\s*/g, '')
    .replace(/<link rel="canonical"[^>]*>\s*/g, '')
    .replace('</head>', () => `  ${head}\n</head>`)
    .replace(ROOT_PLACEHOLDER, () => `<div id="root">${html}</div>`)
}

export function routeToFile(route) {
  return `${route.replace(/^\/+|\/+$/g, '')}/index.html`
}

export function addToSitemap(xml, entries) {
  if (!xml.includes('</urlset>')) throw new Error('sitemap has no </urlset>')
  const block = entries
    .map(
      (e) =>
        `  <url>\n    <loc>${e.loc}</loc>\n    <lastmod>${e.lastmod}</lastmod>\n    <changefreq>${e.changefreq}</changefreq>\n    <priority>${e.priority}</priority>\n  </url>\n`,
    )
    .join('\n')
  return xml.replace('</urlset>', () => `${block}\n</urlset>`)
}

export function addToLlms(txt, items) {
  const lines = items.map((a) => `- [${a.title}](${a.url}): ${a.description}`)
  return `${txt.trimEnd()}\n\n## Blog\n\n${lines.join('\n')}\n`
}
```

import { renderToString } from 'react-dom/server'
import { StaticRouter } from 'react-router-dom/server'
import { Routes, Route } from 'react-router-dom'
import { HelmetProvider, type HelmetServerState } from 'react-helmet-async'
import { LanguageProvider, type Language } from '@/hooks/useLanguage'
import BlogIndex from '@/pages/BlogIndex'
import BlogPost from '@/pages/BlogPost'

export { articles } from '@/content/blog'
export { validateArticles } from '@/content/blog/validate'

export function render(url: string, lang: Language): { html: string; head: string } {
  const helmetContext: { helmet?: HelmetServerState } = {}
  const html = renderToString(
    <HelmetProvider context={helmetContext}>
      <LanguageProvider initialLanguage={lang}>
        <StaticRouter location={url}>
          <Routes>
            <Route path="/blog" element={<BlogIndex />} />
            <Route path="/blog/:slug" element={<BlogPost />} />
          </Routes>
        </StaticRouter>
      </LanguageProvider>
    </HelmetProvider>,
  )
  const h = helmetContext.helmet
  if (!h) throw new Error('Helmet context was not populated')
  const head = [h.title, h.meta, h.link, h.script].map((part) => part.toString()).join('\n  ')
  return { html, head }
}
